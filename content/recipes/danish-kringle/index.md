ingredient_groups:
  - ingredients:
      - line: "1 pkg dry yeast"
      - line: "¼ c warm water"
  - name: "In separate bowl mix"
    ingredients:
      - line: "2 c flour"
      - line: "½ c lard"
        note: "No Substitutes"
      - line: "2 T sugar"
      - line: "½ tsp salt"
  - name: "In another bowl mix together"
    ingredients:
      - line: "2 egg yolks"
      - line: "½ c milk"
      - line: "yeast mixture"

steps:
  - "Dissolve 1 pkg dry yeast in ¼ c warm water – Let set 5-6 min"
  - "Mix til crumbly and size of peas –"
  - "Pour in flour mixture mix well – cover and refrigerate overnight."
  - "Roll in about 3-4 different pieces very thin – put filling to ½ in of edge – roll up – put in greased pan"
  - "Cover let raise about 20-25 min"
  - "Bake 375° 20-30 mins."
  - "Cover with cream cheese frosting –"

editor_notes:
  - "For the filling, use canned poppy seed filling from the store. You can mix a little water into the can to loosen it."
  - "The card doesn't give the cream cheese frosting."
