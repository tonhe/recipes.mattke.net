ingredient_groups:
  - name: "Meat and vegetables"
    ingredients:
      - line: "2-3 lbs beef, sliced ¾-1″ thick"
        note: "cheapest cuts"
      - line: "Salt and pepper"
      - line: "Flour, for rolling"
      - line: "3-4 carrots, cut 3-4″"
      - line: "3-4 stalks celery, cut 3-4″ and ½″ across"
      - line: "2 cloves garlic"
      - line: "2 med onions, chopped"
      - line: "½ green pepper"
  - name: "Gravy"
    ingredients:
      - line: "4 c water"
      - line: "2 T beef base"
        alt: "or 5 bouillon cubes"
      - line: "1 can (15 oz) tomato sauce"
      - line: "1 T Kitchen Bouquet"
      - line: "Salt and pepper"
      - line: "Flour and water, to thicken"

steps:
  - "Salt and pepper the beef on each side, roll in flour, and pound to help tenderize. Sear."
  - "**Gravy:** heat the gravy ingredients except for the tomato sauce. Thicken with flour/water when the mixture boils. Add the tomato sauce just before it comes off the stove."
  - "Layer meat, veggies, and gravy in a baking pan. Cover tightly with foil."
  - "Bake at 400° for 1 hour, then turn down to 325° for 3 hours."
