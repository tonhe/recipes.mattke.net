people: ["Tony"]
categories: ["Main dishes"]
tags: ["chicken", "beans", "chili", "soup"]
time: "about 2 hours"
time_label: "Simmer"

ingredients:
  - line: "10 cups cooked Great Northern Beans"
  - line: "2 pounds boneless chicken breast"
  - line: "2 medium onions, chopped"
  - line: "2 ribs celery, chopped"
  - line: "2 4-ounce cans chopped mild green chilis"
  - line: "4 cloves garlic, minced"
  - line: "1 Tbsp. olive oil"
  - line: "6 cups chicken stock or canned broth"
  - line: "2 tsp. ground cumin"
  - line: "1½ tsp. dried oregano, crumbled"
  - line: "¼ tsp. cayenne pepper"
  - line: "½ tsp. black pepper"
  - line: "2 tsp. white pepper"
    editor: "The amount is faded off the page."
  - line: "salt to taste"
  - line: "¼ tsp. ground cloves"
  - line: "3 cups shredded Monterey Jack cheese"
  - line: "chopped fresh cilantro"
  - line: "sour cream"
  - line: "salsa"

steps:
  - "Simmer chicken in large pot about 15 minutes."
  - "Drain, cool, remove skin, and cube."
  - "Heat oil in same pot over medium high heat and sauté onions & celery until translucent."
  - "Stir in garlic, chilies, cumin, oregano, cloves, salt and peppers (red, black, & white) and saute 2 minutes."
  - "Add beans, chicken & stock; bring to boil and reduce heat."
  - "Simmer about 2 hours."
  - "Add 1 cup cheese to chili and stir until cheese melts."
  - "Garnish with remaining cheese, sour cream, salsa, & cilantro."
  - "Serve."
