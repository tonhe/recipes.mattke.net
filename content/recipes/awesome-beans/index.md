---
title: "Awesome Beans"
summary: "Canned baked beans simmered with browned ground beef, onion, garlic, mustard, brown sugar, Worcestershire and BBQ sauce."
people: ["Tony Mattke"]
categories: ["Sides"]
tags: ["beans", "beef"]
time: "30 min"
time_label: "Simmer"

ingredient_groups:
  - ingredients:
      - line: "1lb g. beef"
  - name: "①"
    ingredients:
      - line: "1 sweet onion (minced)"
      - line: "2 cloves garlic (minced)"
  - name: "②"
    ingredients:
      - line: "S+P to taste"
      - line: "½ tsp. garlic powder"
      - line: "½ tsp. onion powder"
  - name: "③"
    ingredients:
      - line: "1 28 oz can of baked beans"
      - line: "1½ tbsp yellow mustard"
      - line: "2 tbsp b. sugar"
      - line: "2 tbsp worcestershire"
      - line: "3/4 C BBQ sauce"

steps:
  - "brown beef – add onion/garlic ①"
  - "saute until onion = transparent (4-5 min)"
  - "add ② – 2 mins"
  - "add beans + rest of ③"
  - "lower heat – simmer 30 min"
