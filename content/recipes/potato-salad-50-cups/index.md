ingredients:
  - line: "15 lbs potatoes (boil with skins on – peel)"
  - line: "4 dz. eggs"
  - line: "1c vinegar"
    note: "try 1 cup then add more if needed"
  - line: "4 T Salt"
  - line: "½ c. dry onion"
  - line: "1 c. prepared mustard"
  - line: "1⅓ c sugar"
  - line: "about 2 qts Miracle Whip –"
  - line: "chopped onion tops"
  - line: "½ small pimiento"
    alt: "or 1 red pepper"

steps:
  - "Cut potatoes – add onions, vinegar, salt – mix well –"
  - "let stand at least an hr. overnight better"
  - "add all ingr. except eggs – mix well –"
  - "add eggs – mix."
  - "Let stand at least an 1 hr before using"
