ingredients:
  - line: "1 lb bacon (1 inch pieces)"
  - line: "4 c chopped celery"
  - line: "1 large chopped onion"
  - line: "7½ c water"
  - line: "2 lb hash browns (Southern style)"
  - line: "2 cans (10 oz) cream of Chicken Soup"
  - line: "2 cans Cheddar Cheese soup"
  - line: "1 lb Velveeta (cubed)"
  - line: "1 lb frozen cauliflower & broccoli"
    note: "If desired add"

steps:
  - "Brown – 1 lb bacon (1 inch pieces) Drain"
  - "Cook – 4 c chopped celery, 1 large chopped onion in 2-3 T bacon grease"
  - "Add 7½ c water"
  - "Add 2 lb hash browns (Southern style)"
  - "Cook til tender"
  - "Pour in slow cooker"
  - "Add 2 cans (10 oz) cream of Chicken Soup, 2 cans Cheddar Cheese soup, 1 lb Velveeta (cubed)"
  - "If desired add 1 lb frozen cauliflower & broccoli"
  - "Cook 8 hrs or nearly"

author_note: "Can use double boiler instead of slow cooker"
