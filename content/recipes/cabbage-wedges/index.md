title: "Cabbage Wedges"
summary: "Cabbage wedges cooked crisp tender, baked under a pepper, onion and butter white sauce, then topped with cheddar, mayo and chili sauce for a few more minutes."
people: ["Grandma Vi"]
categories: ["Sides"]
tags: ["cabbage", "casserole", "cheese"]
oven: "375°"
time: "15 min then 5 min"

ingredient_groups:
  - ingredients:
      - line: "3 lb wedges of cabbage (10-12)"
      - line: "½ c gr pepper"
      - line: "¼ c gr onion"
      - line: "¼ c butter"
      - line: "¼ c flour"
      - line: "½ tsp salt"
      - line: "¼ tsp pepper"
      - line: "2 c milk"
  - name: "Combine"
    ingredients:
      - line: "¾ c cheddar (shredded)"
      - line: "½ c mayo"
      - line: "3 T chili sauce"

steps:
  - "Cook in salt water 10-15 min til crisp tender"
  - "Drain"
  - "Place in greased 3 qt casserole"
  - "Saute ½ c gr pepper, ¼ c gr onion, ¼ c butter"
  - "Then add ¼ c flour, ½ tsp salt, ¼ tsp pepper, 2 c milk"
  - "Pour over cabbage"
  - "Bake 15 min"
  - "Combine ¾ c cheddar (shredded), ½ c mayo, 3 T chili sauce"
  - "Pour over"
  - "Bake another 5 min"
